cmake_minimum_required(VERSION 3.21)

project(BackType VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BACKTYPE_BUILD_AE_PLUGIN "Build the After Effects plugin target" ON)
option(BACKTYPE_BUILD_TESTS "Build platform-neutral tests" ON)

if(BACKTYPE_BUILD_TESTS)
    add_executable(backtype_text_logic_tests
        tests/BackType_TextLogicTests.cpp
        src/BackType_TextLogic.cpp
    )
    target_include_directories(backtype_text_logic_tests PRIVATE src)
endif()

if(NOT BACKTYPE_BUILD_AE_PLUGIN)
    return()
endif()

set(AE_SDK_ROOT "" CACHE PATH "Path to the Adobe After Effects C++ SDK root")
if(NOT AE_SDK_ROOT)
    message(FATAL_ERROR "Set AE_SDK_ROOT to your local Adobe After Effects SDK path.")
endif()

set(BACKTYPE_COMMON_SOURCES
    src/BackType.cpp
    src/BackType_Strings.cpp
    src/BackType_TextLogic.cpp
)

set(BACKTYPE_SDK_INCLUDE_DIRS
    "${AE_SDK_ROOT}/Headers"
    "${AE_SDK_ROOT}/Resources"
    "${AE_SDK_ROOT}/Util"
)

if(WIN32)
    set(AE_PIPL_TOOL "" CACHE FILEPATH "Path to Adobe PiPLTool/cnvtpipl for Windows resource generation")
    set(BACKTYPE_PREBUILT_PIPL_RRC "" CACHE FILEPATH "Optional prebuilt BackType_PiPL.rrc generated by the Adobe SDK tools")
    set(BACKTYPE_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
    file(MAKE_DIRECTORY "${BACKTYPE_GENERATED_DIR}")

    if(BACKTYPE_PREBUILT_PIPL_RRC)
        set(BACKTYPE_PIPL_RRC "${BACKTYPE_PREBUILT_PIPL_RRC}")
    elseif(AE_PIPL_TOOL)
        set(BACKTYPE_PIPL_RRC "${BACKTYPE_GENERATED_DIR}/BackType_PiPL.rrc")
        add_custom_command(
            OUTPUT "${BACKTYPE_PIPL_RRC}"
            COMMAND "${AE_PIPL_TOOL}" "${CMAKE_CURRENT_SOURCE_DIR}/src/BackType_PiPL.r" "${BACKTYPE_PIPL_RRC}"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/BackType_PiPL.r"
            COMMENT "Generating Windows PiPL resource"
            VERBATIM
        )
    else()
        message(FATAL_ERROR "Windows builds need AE_PIPL_TOOL or BACKTYPE_PREBUILT_PIPL_RRC so BackType.aex contains its PiPL.")
    endif()

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/packaging/BackType.rc.in"
        "${BACKTYPE_GENERATED_DIR}/BackType.rc"
        @ONLY
    )

    add_library(BackType MODULE
        ${BACKTYPE_COMMON_SOURCES}
        src/TextRenderer_Windows.cpp
        "${BACKTYPE_GENERATED_DIR}/BackType.rc"
        "${BACKTYPE_PIPL_RRC}"
    )

    target_include_directories(BackType PRIVATE
        src
        ${BACKTYPE_SDK_INCLUDE_DIRS}
    )
    target_compile_definitions(BackType PRIVATE
        WIN_ENV=1
        _WINDOWS=1
        NOMINMAX=1
    )
    target_link_libraries(BackType PRIVATE gdiplus)
    set_target_properties(BackType PROPERTIES
        PREFIX ""
        OUTPUT_NAME "BackType"
        SUFFIX ".aex"
    )

elseif(APPLE)
    enable_language(OBJCXX)

    add_library(BackType MODULE
        ${BACKTYPE_COMMON_SOURCES}
        src/TextRenderer_macOS.mm
    )

    target_include_directories(BackType PRIVATE
        src
        ${BACKTYPE_SDK_INCLUDE_DIRS}
    )
    target_compile_definitions(BackType PRIVATE
        MAC_ENV=1
    )
    target_link_libraries(BackType PRIVATE
        "-framework CoreFoundation"
        "-framework CoreGraphics"
        "-framework CoreText"
    )
    set_target_properties(BackType PROPERTIES
        PREFIX ""
        OUTPUT_NAME "BackType"
        BUNDLE TRUE
        BUNDLE_EXTENSION "plugin"
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in"
    )

    add_custom_command(TARGET BackType POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:BackType>/Contents/Resources"
        COMMAND /usr/bin/Rez
            -useDF
            -I "${AE_SDK_ROOT}/Headers"
            -I "${AE_SDK_ROOT}/Resources"
            -I "${CMAKE_CURRENT_SOURCE_DIR}/src"
            "${CMAKE_CURRENT_SOURCE_DIR}/src/BackType_PiPL.r"
            -o "$<TARGET_BUNDLE_DIR:BackType>/Contents/Resources/BackType.rsrc"
        COMMENT "Compiling BackType PiPL resource"
    )
else()
    message(FATAL_ERROR "BackType currently has plugin build targets for Windows and macOS only.")
endif()
